// Helper: Generate 5 weeks (current week + next 4)
const generateWeekOptions = () => {
  const options = [];
  const now = new Date();
  const monday = new Date(now);
  const day = monday.getDay();
  const diff = day === 0 ? -6 : 1 - day;
  monday.setDate(monday.getDate() + diff); // move to Monday

  for (let i = 0; i < 5; i++) {
    const weekStart = new Date(monday);
    weekStart.setDate(monday.getDate() + i * 7);
    const weekEnd = new Date(weekStart);
    weekEnd.setDate(weekStart.getDate() + 6);
    const label = `Week of ${weekStart.toLocaleDateString()} – ${weekEnd.toLocaleDateString()}`;
    options.push({ value: weekStart.toISOString(), label });
  }
  return options;
};
