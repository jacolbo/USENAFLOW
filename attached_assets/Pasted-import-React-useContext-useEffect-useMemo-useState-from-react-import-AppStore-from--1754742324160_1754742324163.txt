import React, { useContext, useEffect, useMemo, useState } from 'react';
import { AppStore } from '../state/AppStore';
import { fetchProjects } from '../api';
import { format, startOfDay, startOfWeek, startOfMonth, isWithinInterval, eachDayOfInterval, eachWeekOfInterval, eachMonthOfInterval } from 'date-fns';
import { Line, Bar } from 'react-chartjs-2';
import { Chart, CategoryScale, LinearScale, PointElement, LineElement, BarElement, Tooltip, Legend } from 'chart.js';

Chart.register(CategoryScale, LinearScale, PointElement, LineElement, BarElement, Tooltip, Legend);

type Project = {
  id: string|number;
  status?: string;
  deliveredAt?: string; // ISO
  selectedCount?: number;      // preferred field for total photos retouched
  packageCount?: number;       // fallback inputs if selectedCount absent
  extras?: number;             // fallback inputs if selectedCount absent
  assignedTo?: string;
};

const ROLES_OK = new Set(['Admin','WorkflowManager','Sales']);
const toDate = (v?: string) => v ? new Date(v) : null;
const photosInProject = (p: Project) => {
  if (typeof p.selectedCount === 'number') return Math.max(0, p.selectedCount);
  const pkg = typeof p.packageCount === 'number' ? p.packageCount : 0;
  const ex  = typeof p.extras === 'number' ? p.extras : 0;
  return Math.max(0, pkg + ex);
};

type Granularity = 'daily' | 'weekly' | 'monthly';

export default function TeamProgress(){
  const { user } = useContext(AppStore);
  const [projects, setProjects] = useState<Project[]>([]);
  const [gran, setGran] = useState<Granularity>('daily');
  const [range, setRange] = useState<{from: string; to: string}>(() => {
    const to = new Date();
    const from = new Date(); from.setMonth(to.getMonth()-1);
    return { from: from.toISOString().slice(0,10), to: to.toISOString().slice(0,10) };
  });

  useEffect(()=>{ (async ()=>{
    try {
      const data = await fetchProjects();
      setProjects(Array.isArray(data) ? data : []);
    } catch { setProjects([]); }
  })(); },[]);

  const filtered = useMemo(() => {
    const from = new Date(range.from + 'T00:00:00');
    const to   = new Date(range.to   + 'T23:59:59');
    return projects.filter(p => {
      if (p.status !== 'Delivered') return false;
      const d = toDate(p.deliveredAt);
      if (!d) return false;
      return isWithinInterval(d, { start: from, end: to });
    });
  }, [projects, range.from, range.to]);

  // Build buckets by granularity
  const series = useMemo(() => {
    const from = new Date(range.from + 'T00:00:00');
    const to   = new Date(range.to   + 'T23:59:59');

    let buckets: Date[] = [];
    let labeler: (d: Date)=>string;
    let keyer:   (d: Date)=>number;

    if (gran === 'daily') {
      buckets = eachDayOfInterval({ start: from, end: to }).map(d => startOfDay(d));
      labeler = d => format(d, 'EEE d MMM');
      keyer   = d => startOfDay(d).getTime();
    } else if (gran === 'weekly') {
      buckets = eachWeekOfInterval({ start: from, end: to }, { weekStartsOn: 1 }).map(d => startOfWeek(d, { weekStartsOn: 1 }));
      labeler = d => `Wk ${format(d,'d MMM')}`;
      keyer   = d => startOfWeek(d, { weekStartsOn: 1 }).getTime();
    } else {
      buckets = eachMonthOfInterval({ start: from, end: to }).map(d => startOfMonth(d));
      labeler = d => format(d, 'MMM yyyy');
      keyer   = d => startOfMonth(d).getTime();
    }

    const map = new Map<number, number>(); // key -> photos
    for (const b of buckets) map.set(keyer(b), 0);

    for (const p of filtered) {
      const d = toDate(p.deliveredAt)!;
      const photos = photosInProject(p);
      let bin: Date;
      if (gran === 'daily') bin = startOfDay(d);
      else if (gran === 'weekly') bin = startOfWeek(d, { weekStartsOn: 1 });
      else bin = startOfMonth(d);
      const key = keyer(bin);
      map.set(key, (map.get(key) || 0) + photos);
    }

    const labels = buckets.map(b => labeler(b));
    const values = buckets.map(b => map.get(keyer(b)) || 0);
    return { labels, values };
  }, [filtered, gran, range.from, range.to]);

  if (!user || !ROLES_OK.has(user.role)) {
    return <div style={{opacity:.8}}>You don’t have access to Team Progress.</div>;
  }

  return (
    <div>
      <h2>Team Progress</h2>
      <p style={{opacity:.75, marginTop:-6}}>Photos counted only when Sales/Admin mark the project as <b>Delivered</b>.</p>

      <div style={{display:'flex', gap:12, flexWrap:'wrap', margin:'12px 0'}}>
        <label>From <input type="date" value={range.from} onChange={e=>setRange(r=>({...r, from:e.target.value}))} /></label>
        <label>To <input type="date" value={range.to} onChange={e=>setRange(r=>({...r, to:e.target.value}))} /></label>
        <label>View
          <select value={gran} onChange={e=>setGran(e.target.value as Granularity)} style={{marginLeft:8}}>
            <option value="daily">Daily</option>
            <option value="weekly">Weekly</option>
            <option value="monthly">Monthly</option>
          </select>
        </label>
      </div>

      {/* Line chart: photos over time */}
      <div style={{background:'#121212', border:'1px solid #333', borderRadius:12, padding:12, marginBottom:16}}>
        <h3 style={{margin:'6px 0 12px'}}>Photos Over Time</h3>
        <Line data={{
          labels: series.labels,
          datasets: [{ label: 'Photos', data: series.values }]
        }} options={{
          responsive: true,
          plugins: { legend: { display: false } },
          elements: { point: { radius: 3 } },
          scales: { x: { ticks: { maxRotation:0, autoSkip:true } } }
        }} />
      </div>

      {/* Histogram: distribution of photos per bucket */}
      <div style={{background:'#121212', border:'1px solid #333', borderRadius:12, padding:12}}>
        <h3 style={{margin:'6px 0 12px'}}>Histogram</h3>
        <Bar data={{
          labels: series.labels,
          datasets: [{ label: 'Photos', data: series.values }]
        }} options={{
          responsive: true,
          plugins: { legend: { display: false } },
          scales: { x: { ticks: { maxRotation:0, autoSkip:true } } }
        }} />
      </div>
    </div>
  );
}
